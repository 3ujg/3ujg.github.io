CREATE TABLE users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    first_name TEXT NOT NULL,
    last_name TEXT NOT NULL,
    email TEXT NOT NULL,
    phone TEXT NOT NULL,
    image TEXT
);

Ülesanne 14: CASE lause [cite: 4]

SQL SQLite

Kasutajate sorteerimine ja uue veeru 'Asutus' tekitamine vastavalt e-posti aadressi lõpule[cite: 5, 6, 7, 8, 9, 10, 11, 12, 13].

SELECT
  *,
  CASE
    WHEN email LIKE '%.edu' THEN 'HARIDUSASUTUS'
    WHEN email LIKE '%.gov' THEN 'RIIGIASUTUS'
    ELSE 'TAVAISIK'
  END AS Asutus
FROM users
ORDER BY email;

Ülesanne 15: Andmete uuendamine [cite: 14]

Olemasoleva kirje muutmine andmebaasis ning muudetud andmete kohene tagastamine[cite: 15, 16, 17, 18, 19].

UPDATE users
SET email = 'muudetud@gmail.com',
    phone = '123'
WHERE id = 1
RETURNING id, email, phone;

Ülesanne 16: Tabelite ühendamine (JOIN) [cite: 20]

SQL INNER JOIN

Erinevate tabelite omavaheline sidumine, et pärida seotud andmeid[cite: 21]. Külalised ja nende broneeringud[cite: 22, 23, 24, 25]:

SELECT users.first_name, users.last_name, bookings.rooms_id, bookings.checkin, bookings.checkout
FROM users
INNER JOIN bookings ON users.id = bookings.users_id;

Broneeringute maksete ajalugu[cite: 26, 27, 28, 29]:

SELECT bookings.users_id, payments.payment_date, payments.amount
FROM bookings
INNER JOIN payments ON bookings.users_id = payments.booking_id;

Külalised koos broneeritud tubadega[cite: 30, 31, 32, 33, 34]:

SELECT users.first_name, users.last_name, rooms.room_number
FROM bookings
JOIN users ON users_id = users.id
JOIN rooms ON rooms_id = rooms.id;
